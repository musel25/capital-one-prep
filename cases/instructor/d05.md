```python
def net_by_category(purchases, refunds):
    index = {p["id"]: p for p in purchases}
    totals = {}
    for p in purchases:
        cat = p["category"]
        totals[cat] = totals.get(cat, 0) + p["amount"]
    for r in refunds:
        if r["state"] == "posted":
            cat = index[r["purchase_id"]]["category"]
            totals[cat] -= r["amount"]
    return totals

def refundable(purchases, refunds, purchase_id):
    index = {p["id"]: p for p in purchases}
    if purchase_id not in index:
        raise ValueError("unknown purchase")
    refunded = sum(r["amount"] for r in refunds
                   if r["purchase_id"] == purchase_id and r["state"] == "posted")
    return max(0, index[purchase_id]["amount"] - refunded)
```
