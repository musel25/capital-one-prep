```python
def settle(rows, today, waiting_days):
    cutoff = today - waiting_days
    out = []
    for r in sorted(rows, key=lambda r: r["id"]):
        if r["captured_day"] <= cutoff and not r["held"]:
            fee = r["gross_cents"] * 125 // 10000
            out.append({"id": r["id"], "merchant": r["merchant"],
                        "net_cents": r["gross_cents"] - fee})
    return out

def payouts(settled, reserves):
    totals = {}
    for r in settled:
        merchant = r["merchant"]
        totals[merchant] = totals.get(merchant, 0) + r["net_cents"]
    return {m: max(0, total-reserves.get(m, 0)) for m, total in totals.items()}
```
