```python
def select_alerts(rows):
    latest = {}
    for r in rows:
        if r["state"] != "posted" or r["amount_cents"] < 2000:
            continue
        old = latest.get(r["merchant"])
        if old is None or (r["minute"], r["id"]) > (old["minute"], old["id"]):
            latest[r["merchant"]] = r
    return sorted(latest.values(), key=lambda r: (r["minute"], r["id"]))

def count_by_hour(selected_rows):
    counts = {}
    for r in selected_rows:
        hour = r["minute"] // 60
        counts[hour] = counts.get(hour, 0) + 1
    return counts
```
