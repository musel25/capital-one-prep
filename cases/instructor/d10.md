```python
def reserve(stock, reservations, reservation_id, items):
    if any(stock[product] < quantity for product, quantity in items):
        return False
    for product, quantity in items:
        stock[product] -= quantity
    reservations[reservation_id] = dict(items)
    return True

def cancel(stock, reservations, reservation_id, cancelled):
    if reservation_id not in reservations:
        return "unknown"
    if reservation_id in cancelled:
        return "already_cancelled"
    for product, quantity in reservations[reservation_id].items():
        stock[product] += quantity
    cancelled.add(reservation_id)
    return "cancelled"
```
