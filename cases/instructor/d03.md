```python
def remaining(used, customer, month):
    return max(0, 100 - used.get((customer, month), 0))

def award(rows, used):
    out = []
    for r in rows:
        if r["state"] != "posted":
            continue
        key = (r["customer"], r["month"])
        raw = r["amount_cents"] * 2 // 100
        given = min(raw, remaining(used, *key))
        used[key] = used.get(key, 0) + given
        out.append((r["id"], given))
    return out
```
