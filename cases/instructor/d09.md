```python
def eligible(p):
    return (100 <= p["minute"] < 160
            and p["amount_cents"] >= 5000
            and p["new_customer"] and p["category"] == "travel")

def capped_awards(rows, used):
    new_used = dict(used)
    out = []
    for p in rows:
        if not eligible(p):
            continue
        customer = p["customer"]
        available = max(0, 300-new_used.get(customer, 0))
        reward = min(p["amount_cents"]*5//100, available)
        new_used[customer] = new_used.get(customer, 0)+reward
        out.append((p["id"], reward))
    return out, new_used
```
